1.4 虚函数 vs 非虚函数

虚函数 virtual非虚函数
通过对象调用静态绑定（自身版本）静态绑定
通过指针调用动态绑定（实际类型）静态绑定（指针类型）
通过引用调用动态绑定（实际类型）静态绑定（引用类型）

🔥 虚函数默认参数陷阱

class Animal { virtual void speak(int n = 5) const; };  // 默认值 5
class Dog : public Animal { void speak(int n = 9) const override; }; // 默认值 9

Animal* p = new Dog();
p->speak();  // 函数体用 Dog 的，默认值用 Animal 的！

默认参数是静态绑定的——看声明类型（指针类型），不看实际类型。

📝 第34题 — 纯虚函数与抽象类

class Oyster {
public:
    Oyster(string genusString) { genus = genusString; }
    string getPhylum() const { return "Mollusca"; }
    virtual string getName() const { return "Oyster class"; }
    string getGenus() const { return genus; }
    virtual void print() const = 0;   // ← 纯虚函数！Oyster 是抽象类
private:
    string genus;
};
class VirginiaOyster : public Oyster {
public:
    VirginiaOyster() : Oyster("Crassostrea") {}
    virtual string getName() const { return "VirginiaOyster class"; }
    virtual void print() const {
        cout << "Phylum:" << getPhylum().c_str()
             << "\tGenus: " << getGenus().c_str()
             << "\tName: " << getName().c_str();
    }
};
int main() {
    VirginiaOyster oyster;
    Oyster *baseClassPtr = &oyster;  // 基类指针 → 派生类对象
    baseClassPtr->print();             // 动态绑定！
}
🔍 点击查看输出与解析

输出：Phylum:Mollusca    Genus: Crassostrea    Name: VirginiaOyster class

  • print() 是纯虚函数 → 必须由派生类实现
  • getName() 是虚函数 → 动态绑定到 VirginiaOyster::getName()

2.4 String 类完整实现 —— 必背！

⚠️ 这是每年必考的手写代码，必须能默写！

class String {
private:
    char *m_data;
public:
    // ① 普通构造函数
    String(const char *str = NULL) {
        if (str == NULL) {
            m_data = new char[1];
            m_data[0] = '\0';
        } else {
            m_data = new char[strlen(str) + 1];   // +1 for '\0'
            strcpy(m_data, str);
        }
    }

    // ② 拷贝构造函数（深拷贝）
    String(const String &other) {
        m_data = new char[strlen(other.m_data) + 1];
        strcpy(m_data, other.m_data);
    }

    // ③ 赋值运算符（三步走）
    String& operator=(const String &other) {
        if (this == &other) return *this;    // 一看：检查自赋值
        delete[] m_data;                       // 二清：释放旧内存
        m_data = new char[strlen(other.m_data) + 1]; // 三拷：分配+拷贝
        strcpy(m_data, other.m_data);
        return *this;                          // 四返
    }

    // ④ 析构函数
    ~String() { delete[] m_data; }
};
📝 赋值运算符口诀：一看（自赋值检查）→ 二清（释放旧内存）→ 三拷（分配+拷贝）→ 四返（return *this）

为什么拷贝构造函数必须用引用传参？

// ❌ 错误！导致无限递归
String(const String other) { ... }
// 按值传递 → 形参初始化又要调拷贝构造 → 死循环！

// ✅ 正确
String(const String &other) { ... }

4.1 第3题：输入验证 + 求平方

#include <iostream>
#include <string>
#include <cctype>
using namespace std;

int main() {
    string input;
    cout << "请输入一个整数：";
    cin >> input;

    if (input.empty()) {
        cout << "错误：输入为空！" << endl;
        return -1;
    }

    int start = (input[0] == '-') ? 1 : 0;   // 处理负号
    if (start == 1 && input.length() == 1) {
        cout << "错误：请输入有效整数！" << endl;
        return -1;
    }

    for (int i = start; i < input.length(); i++) {
        if (!isdigit(input[i])) {
            cout << "错误：包含非数字字符！" << endl;
            return -1;
        }
    }

    int num = stoi(input);
    cout << num << " 的平方是：" << num * num << endl;
    return 0;
}

4.4 第43题：Stack 类模板

template <typename T>
class Stack {
    T* data;
    int top;        // 空栈 = -1
    int capacity;

    void resize() {
        capacity *= 2;
        T* newData = new T[capacity];
        for (int i = 0; i <= top; i++) newData[i] = data[i];
        delete[] data;
        data = newData;
    }

public:
    Stack(int cap = 8) : top(-1), capacity(cap) { data = new T[capacity]; }
    ~Stack() { delete[] data; }

    void push(T value) {
        if (top + 1 >= capacity) resize();
        data[++top] = value;
    }

    T pop() {
        if (top < 0) throw underflow_error("Stack is empty");
        return data[top--];
    }

    T peek() const {
        if (top < 0) throw underflow_error("Stack is empty");
        return data[top];
    }

    int getSize() const { return top + 1; }

    friend ostream& operator<<(ostream& os, const Stack<T>& s) {
        for (int i = s.top; i >= 0; i--) {
            os << s.data[i];
            if (i > 0) os << " ";
        }
        return os;
    }
};
📦 入栈 push(42)：
top=-1 → data[0]=42, top=0
📤 出栈 pop()：
返回 data[top], top-- → 返回 42, top=-1
🔍 查看 peek()：
返回 data[top] 但不修改 top
📏 扩容 resize()：
top + 1 >= capacity 时触发，容量翻倍

public:
    Resource(const char* str) {
        buffer = new char[strlen(str)];
        strcpy(buffer, str);
    }
    ~Resource() { delete buffer; }
};
int main() {
    Resource a("hello");
    Resource b = a;   // ①
    b = a;            // ②
}
